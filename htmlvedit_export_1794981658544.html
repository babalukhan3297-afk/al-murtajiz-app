import { useServerFn } from "@tanstack/react-start"; import { toast } from "sonner"; import { submitToSheet, type SheetAction } from "./sheets.functions"; /** Sends a form to the Google Sheet and tells the user what happened. */ export function useSheetSubmit() { const send = useServerFn(submitToSheet); return async (data: SheetAction, successMsg: string) => { try { const res = await send({ data }); if (res.ok) { toast.success(successMsg); } else if (res.reason === "not_configured") { toast.warning("Saved on this phone", { description: "Your Google Sheet isn't accepting submissions yet. It will sync once the sheet is set up.", }); } else { toast.warning("Saved on this phone", { description: "Couldn't reach the Google Sheet. Please try again later.", }); } return res.ok; } catch { toast.error("Please check the details and try again."); return false; } }; }
